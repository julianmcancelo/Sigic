-- ============================================================================

-- 1. EXTENSIONES Y ENUMS
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";
CREATE EXTENSION IF NOT EXISTS "pgcrypto";

CREATE TYPE rol_usuario AS ENUM ('ADMIN', 'ADMINISTRATIVO', 'PORTERIA');
CREATE TYPE estado_egresado AS ENUM ('PENDIENTE', 'ACEPTADO', 'RECHAZADO');
CREATE TYPE tipo_padrino AS ENUM ('PROFESOR', 'FAMILIAR', 'EXTERNO');
CREATE TYPE resultado_otp AS ENUM ('ENVIADO', 'VERIFICADO', 'FALLIDO', 'EXPIRADO');
CREATE TYPE tipo_log AS ENUM ('INSERT', 'UPDATE', 'DELETE', 'LOGIN', 'LOGOUT');
CREATE TYPE tipo_asiento AS ENUM ('EGRESADO', 'INVITADO', 'PADRINO', 'RESERVA');

-- 2. FUNCIONES Y TRIGGERS
CREATE OR REPLACE FUNCTION fn_actualizar_timestamp()
RETURNS TRIGGER AS $$
BEGIN
    NEW.actualizado_en = CURRENT_TIMESTAMP;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- 3. TABLAS CON CLAVES PRIMARIAS Y FORANEAS
CREATE TABLE usuarios (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    nombre VARCHAR(150) NOT NULL,
    email VARCHAR(150) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    rol rol_usuario NOT NULL DEFAULT 'ADMINISTRATIVO',
    activo BOOLEAN NOT NULL DEFAULT TRUE,
    creado_en TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    actualizado_en TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE ceremonias (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    nombre VARCHAR(200) NOT NULL,
    fecha DATE NOT NULL,
    lugar VARCHAR(200) NOT NULL DEFAULT 'Sede Beltran',
    max_invitados INT NOT NULL DEFAULT 4 CHECK (max_invitados >= 0),
    max_padrinos INT NOT NULL DEFAULT 3 CHECK (max_padrinos >= 0),
    activa BOOLEAN NOT NULL DEFAULT FALSE,
    creado_en TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);
CREATE UNIQUE INDEX idx_ceremonias_activa_unica ON ceremonias ((activa)) WHERE activa = TRUE;

CREATE TABLE profesores (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    nombre VARCHAR(150) NOT NULL,
    dni VARCHAR(20) NOT NULL UNIQUE,
    materia VARCHAR(150) NOT NULL,
    activo BOOLEAN NOT NULL DEFAULT TRUE,
    creado_en TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE egresados (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    ceremonia_id UUID NOT NULL REFERENCES ceremonias(id) ON DELETE CASCADE,
    token VARCHAR(32) NOT NULL UNIQUE,
    nombre VARCHAR(150) NOT NULL,
    legajo VARCHAR(50) NOT NULL,
    dni VARCHAR(20) NOT NULL,
    correo VARCHAR(150) NOT NULL,
    telefono VARCHAR(50),
    estado estado_egresado NOT NULL DEFAULT 'PENDIENTE',
    promedio NUMERIC(4,2) CHECK (promedio >= 0 AND promedio <= 10),
    asiento_id VARCHAR(30),
    creado_en TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT uq_egresado_ceremonia_dni UNIQUE (ceremonia_id, dni)
);

CREATE TABLE invitados (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    egresado_id UUID NOT NULL REFERENCES egresados(id) ON DELETE CASCADE,
    nombre VARCHAR(150) NOT NULL,
    dni VARCHAR(20) NOT NULL,
    telefono VARCHAR(50),
    correo VARCHAR(150),
    relacion VARCHAR(80),
    asiento_id VARCHAR(30),
    discapacidad BOOLEAN NOT NULL DEFAULT FALSE,
    presente BOOLEAN NOT NULL DEFAULT FALSE,
    fecha_presente TIMESTAMPTZ,
    creado_en TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT uq_invitado_egresado_dni UNIQUE (egresado_id, dni)
);

CREATE TABLE padrinos (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    egresado_id UUID NOT NULL REFERENCES egresados(id) ON DELETE CASCADE,
    tipo tipo_padrino NOT NULL,
    profesor_id UUID REFERENCES profesores(id) ON DELETE SET NULL,
    invitado_id UUID REFERENCES invitados(id) ON DELETE SET NULL,
    nombre VARCHAR(150),
    asiento_id VARCHAR(30),
    orden INT NOT NULL DEFAULT 1 CHECK (orden > 0),
    creado_en TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT uq_padrino_orden UNIQUE (egresado_id, orden),
    CONSTRAINT ck_padrino_consistencia CHECK (
        (tipo = 'PROFESOR' AND profesor_id IS NOT NULL) OR
        (tipo = 'FAMILIAR' AND invitado_id IS NOT NULL) OR
        (tipo = 'EXTERNO'  AND nombre IS NOT NULL)
    )
);

CREATE TABLE otp_historial (
    id BIGSERIAL PRIMARY KEY,
    egresado_id UUID NOT NULL REFERENCES egresados(id) ON DELETE CASCADE,
    otp_hash VARCHAR(64) NOT NULL,
    ip_origen VARCHAR(45),
    resultado resultado_otp NOT NULL DEFAULT 'ENVIADO',
    solicitado_en TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    verificado_en TIMESTAMPTZ
);

CREATE TABLE configuracion_anfiteatro (
    id BIGSERIAL PRIMARY KEY,
    ceremonia_id UUID NOT NULL REFERENCES ceremonias(id) ON DELETE CASCADE,
    estructura JSONB NOT NULL DEFAULT '{}'::jsonb,
    mapa_roles JSONB NOT NULL DEFAULT '{}'::jsonb,
    actualizado_en TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE configuracion_sistema (
    clave VARCHAR(100) PRIMARY KEY,
    valor TEXT NOT NULL,
    descripcion TEXT,
    actualizado_en TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE logs_auditoria (
    id BIGSERIAL PRIMARY KEY,
    usuario_id UUID REFERENCES usuarios(id) ON DELETE SET NULL,
    accion tipo_log NOT NULL,
    tabla_afectada VARCHAR(60) NOT NULL,
    registro_afectado UUID,
    valores_anteriores JSONB,
    valores_nuevos JSONB,
    ip_origen VARCHAR(45),
    fecha TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

-- 4. INDICES
CREATE INDEX idx_egresados_token ON egresados(token);
CREATE INDEX idx_egresados_dni ON egresados(dni);
CREATE INDEX idx_egresados_ceremonia ON egresados(ceremonia_id);
CREATE INDEX idx_invitados_egresado ON invitados(egresado_id);
CREATE INDEX idx_invitados_presente ON invitados(presente);
CREATE INDEX idx_padrinos_egresado ON padrinos(egresado_id);
CREATE INDEX idx_otp_egresado ON otp_historial(egresado_id);
CREATE INDEX idx_logs_fecha ON logs_auditoria(fecha DESC);

Update OTP 4. Validación de Código OTP Efímero con Expiración Estricta de 10 Minutos
Comprueba el hash SHA-256 del código efímero emitido, rechazando intentos fuera de la ventana de 10 minutos. 
UPDATE otp_historial
SET resultado = 'VERIFICADO', verificado_en = CURRENT_TIMESTAMP
WHERE id = (
    SELECT id FROM otp_historial
    WHERE egresado_id = 'cccccccc-cccc-4ccc-8ccc-cccccccccccc'
      AND resultado = 'ENVIADO'
      AND otp_hash = encode(digest('849201', 'sha256'), 'hex')
      AND solicitado_en >= (CURRENT_TIMESTAMP - INTERVAL '10 minutes')
    ORDER BY solicitado_en DESC
    LIMIT 1
)
RETURNING id, resultado, verificado_en;
Read Analítico 5. Reporte de Estrado para Ceremonial y Locución (JOIN Polimórfico)
Extrae la nómina de graduados ordenada por mérito académico junto a sus padrinos o profesores entregadores. 
SELECT 
    e.nombre AS egresado,
    e.promedio,
    e.asiento_id AS butaca,
    p.orden,
    CASE 
        WHEN p.tipo = 'PROFESOR' THEN prof.nombre || ' (Docente titular - ' || prof.materia || ')'
        WHEN p.tipo = 'FAMILIAR' THEN inv.nombre || ' (Acompañante - ' || inv.relacion || ')'
        WHEN p.tipo = 'EXTERNO'  THEN p.nombre || ' (Allegado designado)'
    END AS entregador_diploma
FROM egresados e
JOIN padrinos p ON p.egresado_id = e.id
LEFT JOIN profesores prof ON prof.id = p.profesor_id
LEFT JOIN invitados inv ON inv.id = p.invitado_id
WHERE e.ceremonia_id = 'aaaaaaaa-aaaa-4aaa-8aaa-aaaaaaaaaaaa'
  AND e.estado = 'ACEPTADO'
ORDER BY e.promedio DESC, e.nombre ASC, p.orden ASC;

5. Glosario de Términos (En Palabras Simples) Conceptos clave explicados de forma clara, directa y sin rodeos
Base de Datos Relacional General
Es un sistema para guardar información en tablas conectadas entre sí mediante enlaces lógicos. Permite que un dato se cargue una sola vez y no se repita en varios lugares.
Clave Primaria (PK) Identificador
Es el código único que identifica a cada fila de una tabla. En SiGIC usamos códigos largos (UUID) para que no haya dos egresados, ceremonias ni usuarios con el mismo código.
Clave Foránea (FK) Relación
Es el puente que conecta dos tablas. Por ejemplo, en la tabla de invitados guardamos a qué egresado acompaña cada persona para saber a quién pertenece cada ingreso.
UUID (Identificador Único) Seguridad
Es un código alfanumérico generado al azar de 36 caracteres. Sirve para que nadie pueda adivinar el enlace privado ni el código QR de otro graduado cambiando simplemente un número.
DDL (Data Definition Language) Estructura
Son las órdenes en SQL que construyen la base: crear las tablas, definir qué tipo de información va en cada columna y aplicar las reglas de seguridad.
CRUD (Operaciones Básicas) Uso Diario
Son las cuatro tareas que hace el sistema: Crear datos (Insert), Leerlos para mostrarlos (Select), Modificarlos cuando hay cambios (Update) y Borrarlos si se cancelan (Delete).
OTP (One-Time Password) Acceso
Código numérico de 6 dígitos que se genera al momento y sirve una sola vez. Llega al correo del graduado para que ingrese sin necesidad de recordar una contraseña fija, y caduca en 10 minutos.
Trigger (Disparador) Automatización
Es una rutina que se ejecuta sola dentro de la base de datos cuando ocurre un evento. En SiGIC se usa para actualizar la fecha y hora exacta cada vez que se modifica un registro.
JSONB Almacenamiento
Es un formato de texto estructurado muy rápido que permite guardar listas complejas dentro de un solo campo, como la distribución de filas y butacas del anfiteatro Beltrán.
Índice B-Tree Rendimiento
Es como el índice de un libro al final: le indica a PostgreSQL exactamente en qué página está el graduado por su DNI o token, evitando tener que revisar toda la tabla uno por uno.
3FN (Tercera Forma Normal) Diseño
Regla de diseño que exige que cada dato dependa únicamente de su clave primaria. Si un egresado cambia su teléfono, se actualiza en su ficha y se refleja en todo el sistema.
Modo Mantenimiento Operación
Ajuste global guardado en la base de datos que suspende temporalmente el acceso público de graduados mientras se realizan tareas técnicas, sin frenar el panel de administración.
